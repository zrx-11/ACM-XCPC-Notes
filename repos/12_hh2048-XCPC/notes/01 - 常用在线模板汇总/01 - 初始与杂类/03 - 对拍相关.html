C++ 对拍

#include <bits/stdc++.h>
using namespace std;
int main() {
    for (int Case = 1; Case <= 10000; Case ++) {
        cout << "==============\n";
        system("Juruo.exe");
        double start = clock();
        system("Shenben.exe");
        double end = clock();
        system("Ask.exe");

        if (system("fc Juruo.out Shenben.out")) {
            cout << "\033[31m" << "Wrong Answer\n" << "\033[0m";
            exit(0);
        } else {
            cout<< "\033[32m" << "Accept!\n"<< "\033[0m"; 
            cout << "测试点 #" << Case << "，用时"  << end - start << "ms\n";
        }
    }
    return 0;
}
